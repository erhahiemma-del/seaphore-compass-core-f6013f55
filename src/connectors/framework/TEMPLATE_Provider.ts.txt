  // ── 2. healthCheck() ──────────────────────────────────────────────
  async healthCheck(): Promise<ConnectorHealth> {
    return {
      connectorId: this.id,
      available: TEMPLATE_PROVIDER_METADATA.enabled,
      authenticated: await this.authenticate(),
      latencyMsP50: 0,
      failureRate: 0,
      quotaRemaining: null,
      lastSuccessAt: null,
      lastError: null,
    };
  }

  // ── 3. search() ───────────────────────────────────────────────────
  async search(query: EvidenceQuery): Promise<ConnectorResult> {
    const key = stableHash({ id: this.id, query });
    const cached = this.cache.get(key);
    if (cached) return cached;

    const startedAt = Date.now();
    let raw: ReadonlyArray<RawRecord> = [];
    let ok = true;
    let error: string | null = null;
    try {
      raw = await this.fetchRaw(query);
    } catch (e) {
      ok = false;
      error = e instanceof Error ? e.message : String(e);
    }

    const records = raw
      .map((r) => this.normalize(r))
      .filter((r): r is NormalizedEvidence => r !== null);

    const result: ConnectorResult = {
      connectorId: this.id,
      ok,
      records,
      latencyMs: Date.now() - startedAt,
      error,
      // Validation FLAGS quality problems; it never drops evidence.
      validation: this.validate(records).issues,
    };
    if (ok) this.cache.set(key, result);
    return result;
  }

  async lookup(query: EvidenceQuery): Promise<ConnectorResult> {
    return this.search(query);
  }

  // ── 4. normalize() ────────────────────────────────────────────────

  }

  // ── 5. validate() ─────────────────────────────────────────────────
  validate(records: ReadonlyArray<NormalizedEvidence>): ProviderValidation {
    return validateRecords(records);
  }
